\section*{Parte A -- Lógica Proposicional Booleana}

\Qitem{ \Qq{Julgue a afirmativa a seguir como \textbf{verdadeira} ou \textbf{falsa}:
    ``Afirmar que uma proposição $p$ é, ao mesmo tempo, verdadeira e falsa, violaria o Princípio do Terceiro Excluído''.
} }

\begin{solution}
\textbf{Falsa}. A afirmativa viola o \textbf{Princípio da Não-Contradição}: uma proposição não pode ser, ao mesmo tempo, verdadeira e falsa. O \textbf{Princípio do Terceiro Excluído} afirma que toda proposição ou é verdadeira ou é falsa, sem uma terceira possibilidade.
\end{solution}
% F. Feedback: O princípio violado seria o Princípio da Não-Contradição.

\Qitem{ \Qq{Na canção Maricotinha, Dorival Caymmi afirma que ``Se fizer bom tempo amanhã eu vou''. Assumindo que esta proposição é verdadeira, assinale a única situação impossível: }
\begin{Qlist}
\item Fez bom tempo e ele foi
\item Fez bom tempo e ele não foi
\item Não fez bom tempo e ele foi
\item Não fez bom tempo e ele não foi
\end{Qlist}
}

\begin{solution}
\textbf{Letra B}: ``Fez bom tempo e ele não foi''. A condicional $p \to q$ (``Se fizer bom tempo, eu vou'') só é falsa quando o antecedente $p$ é verdadeiro e o consequente $q$ é falso. As demais combinações tornam a proposição verdadeira.
\end{solution}
% B. Feedback: A condicional só é falsa caso o antecedente seja verdadeiro e o consequente falso.

\Qitem{ \Qq{Qual é o menor número natural $x$ que torna a sentença aberta ``$x/2 > \pi$'' verdadeira? } }

\begin{solution}
\[
    \frac{x}{2} > \pi \iff x > 2\pi .
\]
Como $2\pi \approx 6{,}28$, o menor número natural que torna a sentença verdadeira é $x = 7$.
\end{solution}
% Resposta: x = 7

% Feedback: $x$ tem que ser maior do que $2\pi$ e  $2\pi < 7$.

\Qitem{ \Qq{Considere os seguintes fatos:}
\inputcode{prolog}{codes/fatos.pl}
Determine o resultado das seguintes consultas. Caso o resultado seja verdadeiro, indique também o valor das variáveis lógicas, quando for o caso.
\begin{Qlist}
\item[(a)] \mintinline{prolog}{?- p(b, a).}
\item[(b)] \mintinline{prolog}{?- p(a, b).}
\item[(c)] \mintinline{prolog}{?- g(X).}
\item[(d)] \mintinline{prolog}{?- p(a, Y).}
\end{Qlist}
}

\begin{solution}
\textbf{(a)} \mintinline{prolog}{?- p(b, a).} $\Rightarrow$ \textbf{falso}: não há fato \mintinline{prolog}{p(b, a)} (o fato \mintinline{prolog}{p(b)} tem aridade $1$).

\textbf{(b)} \mintinline{prolog}{?- p(a, b).} $\Rightarrow$ \textbf{verdadeiro}: corresponde ao fato \mintinline{prolog}{p(a, b)}.

\textbf{(c)} \mintinline{prolog}{?- g(X).} $\Rightarrow$ \textbf{verdadeiro}, com $X = a$ e $X = b$ (fatos \mintinline{prolog}{g(a)} e \mintinline{prolog}{g(b)}; o fato \mintinline{prolog}{g(a, b)} tem aridade $2$ e não unifica).

\textbf{(d)} \mintinline{prolog}{?- p(a, Y).} $\Rightarrow$ \textbf{verdadeiro}, com $Y = b$ e $Y = c$ (fatos \mintinline{prolog}{p(a, b)} e \mintinline{prolog}{p(a, c)}).
\end{solution}

\Qitem{ \Qq{Considere uma consulta composta de Prolog e as quatro portas que controlam o fluxo da execução. Descreva o cenário no qual a saída da consulta exibirá as variáveis lógicas e seus respectivos valores. Explique quando é possível inserir ponto-e-vírgula após o resultado de uma consulta e qual é o significado desta inserção, em termos do fluxo de controle.} }

\begin{solution}
As quatro portas do fluxo de execução são \textbf{call} (chamada do predicado), \textbf{exit} (sucesso da cláusula), \textbf{redo} (tentativa de obter uma nova solução, por retrocesso) e \textbf{fail} (fracasso da cláusula).

A saída exibe as variáveis lógicas e seus valores quando a consulta \textbf{sai} pela porta \textbf{exit}: ou seja, quando uma cláusula é satisfeita e o Prolog mostra a substituição (as ligações) das variáveis que tornam a consulta verdadeira.

O ponto-e-vírgula (\mintinline{prolog}{;}) pode ser digitado sempre que o Prolog exibe uma solução e fica aguardando confirmação. Ao digitá-lo, o usuário aciona a porta \textbf{redo}: o Prolog desfaz as últimas ligações e tenta, por retrocesso (\textit{backtracking}), encontrar a próxima solução. Quando não há mais soluções, a consulta termina em \textbf{fail} e o Prolog responde \mintinline{prolog}{false}.
\end{solution}

\section*{Parte D - Listas e Unificação}

\Qitem{ \Qq{O predicado \mintinline{prolog}{is_set/1} recebe como parâmetro uma lista de inteiros e retorna verdadeiro se todos os elemntos são únicos. Descreva o comportamento da consulta

    \vspace{0.1in}
    \hspace{0.3in}\mintinline{prolog}{?- is_set(1, 2, 3).}
    \vspace{0.1in}
} }

\begin{solution}
O predicado foi declarado como \mintinline{prolog}{is_set/1}:ele recebe \textbf{um único} argumento (uma lista).

A consulta \mintinline{prolog}{?- is_set(1, 2, 3).} tenta chamar \mintinline{prolog}{is_set/3}, que não está definido. Portanto, o Prolog reporta um \textbf{erro} de predicado inexistente (erro de aridade, ``Unknown procedure''). Para funcionar, a consulta deveria passar uma lista, por exemplo \mintinline{prolog}{?- is_set([1, 2, 3]).}
\end{solution}

\Qitem{ \Qq{Determine, sem executar, o resultado da consulta:

    \vspace{0.1in}
    \hspace{0.3in}\mintinline{prolog}{?- X = Y, Y \= Z, Z = W, X = a, W = b.}
    \vspace{0.1in}

Execute esta consulta no \textit{listener} Prolog e compare com sua expectativa. Caso sua expectativa seja diferente do resultado, como a consulta deve ser modificada para obter o retorno esperado?
} }

\begin{solution}
A consulta \textbf{falha} (retorna \mintinline{prolog}{false}). Analisando passo a passo:
\begin{itemize}
    \item \mintinline{prolog}{X = Y}: $X$ e $Y$ passam a compartilhar a mesma variável (ambos livres).
    \item \mintinline{prolog}{Y \= Z}: como $Y$ e $Z$ são livres, elas \textbf{são} unificáveis; logo a desigualdade \mintinline{prolog}{\=} falha e a conjunção termina.
\end{itemize}
Como a segunda meta já falha, o restante (\mintinline{prolog}{Z = W}, \mintinline{prolog}{X = a}, \mintinline{prolog}{W = b}) nunca chega a ser avaliado.

Para obter o sucesso esperado, basta instanciar $Y$ e $Z$ com valores distintos \textbf{antes} de testar a desigualdade, reordenando as metas. Por exemplo, a consulta
\mintinline{prolog}{?- X = Y, Z = W, X = a, W = b, Y \= Z.}
retorna \textbf{verdadeiro}, com $X = Y = a$ e $Z = W = b$.
\end{solution}
% Feedback: $f(0) = f(2\pi)$ e $\f(\mathbb{R}) = [-1, 1]$.
